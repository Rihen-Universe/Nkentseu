  left=2pt, right=2pt, top=4pt, bottom=4pt,
  % \NoAutoSpacing : le francais insere une espace fine avant « : », ce qui

  coltitle=rihenTeal, colbacktitle=rihenTealPale,
  fonttitle=\footnotesize, titlerule=0pt,

  boxed title style={colframe=codeBorder, arc=1pt, boxrule=0.4pt},
  #1}

% ── LE PANNEAU D'ALERTE ──────────────────────────────────────────────────────
% ⚠️ MESURE DU 2026-09-02 : le caractere U+26A0 « ⚠ » NE S'IMPRIME PAS. Ni
% Segoe UI ni Consolas ne le portent dans cette installation, et XeLaTeX le
% laisse simplement TOMBER -- 12 panneaux absents du PDF de ce cours, 17 de
% celui de Jenga. Le journal le dit (`Missing character: There is no ⚠`), mais
% la compilation reste VERTE et le PDF a l'air correct : un paragraphe
% d'avertissement perdait son marqueur sans que rien ne le signale.
%
% ⚠️ Et le selecteur de variante U+FE0F -- la moitie invisible de « ⚠️ » --
% manque AUSSI, ce qui double le compte du journal sans rien ajouter au defaut.
% Compter les lignes « Missing character » surestime donc le nombre de glyphes
% perdus d'un facteur deux exactement.
%
% On le DESSINE plutot que de le composer : aucune dependance de police, meme
% rendu sur toute machine, et il suit la couleur de la charte.
\DeclareRobustCommand{\alerte}{%
  \raisebox{-0.18ex}{\begin{tikzpicture}[scale=0.105, baseline=0]
    \fill[rihenOrange, rounded corners=1.2] (0,0) -- (4.6,8.2) -- (9.2,0) -- cycle;
    \fill[white] (4.05,2.35) rectangle (5.15,6.0);
    \fill[white] (4.05,1.0)  rectangle (5.15,1.95);
  \end{tikzpicture}}\,}

% ── TABLE DES MATIERES ───────────────────────────────────────────────────────
% ⚠️ LA LARGEUR DE LA BOITE A NUMERO SE DIMENSIONNE SUR LE PLUS LONG, PAS SUR
% LE PREMIER. `\contentslabel{L}` ne pousse RIEN : si le numero depasse L, il
% ECRASE le titre au lieu de le decaler -- et le defaut n'apparait qu'a partir
% du chapitre 10, donc jamais pendant qu'on ecrit les premiers.
%   chapitre  : « 26 »    -> 2.0em
%   section   : « 26.14 » -> 3.2em, et l'indentation vaut la boite du chapitre
%                            pour que les deux colonnes de numeros s'alignent.
% Mesure : 27 chapitres et 233 sections dans cette table ; le plus long libelle
% de numero est « 26.14 », c'est lui qui fixe les deux valeurs.
\titlecontents{chapter}[0pt]
  {\addvspace{9pt}\sffamily\bfseries\color{rihenTeal}}
  {\contentslabel[\thecontentslabel]{2.0em}}{}
  {\titlerule*[0.7pc]{.}\contentspage}
\titlecontents{section}[2.0em]
  {\addvspace{2pt}\sffamily\small\color{grisTexte}}
  {\contentslabel{3.2em}}{}
  {\titlerule*[0.7pc]{.}\small\contentspage}
% ⚠️ LES SOUS-SECTIONS AUSSI SONT DANS CETTE TABLE (`tocdepth` vaut 2), et
% c'est LA que le chevauchement se voyait : « 14.11.1Pourquoi chaque dependance
% est la », numero colle au titre. La boite du format par defaut de la classe
% tient « 1.2.3 » et pas « 26.14.1 » -- sept caracteres.
% Mesure : le plus long numero de sous-section de cette table est « 26.14.1 ».
% Indentation = boite chapitre (2.0) + boite section (3.2), pour que les trois
% colonnes de numeros s'alignent en escalier.
\titlecontents{subsection}[5.2em]
  {\addvspace{1pt}\sffamily\small\color{grisTexte}}
  {\contentslabel{4.0em}}{}
  {\titlerule*[0.7pc]{.}\small\contentspage}

% LE CODE EN LIGNE N'EST PAS DU FRANCAIS. La typographie francaise colle une
% espace fine INSECABLE avant « ; : ! ? » -- juste pour du texte, desastreux
% dans \texttt : « input.wheel = 0.f; input.wheelH = 0.f; » devenait UN bloc
% incassable qui debordait de 3 cm dans la marge. \NoAutoSpacing dans \texttt
% rend au code sa ponctuation exacte ET ses points de coupure.
% Robuste, pas fragile : \NoAutoSpacing developpe dans un TITRE de section
% s'ecrit tel quel dans la table des matieres et y explose. La version robuste
% traverse les arguments mobiles sans se developper.
\let\nkOldTexttt\texttt
\DeclareRobustCommand{\nkttCode}[1]{{\NoAutoSpacing\nkOldTexttt{#1}}}
\renewcommand{\texttt}{\nkttCode}
